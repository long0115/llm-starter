**1. `count_tokens` 方法的潜在问题**

```python
# 当前实现（llm_client.py 第32-36行）
def count_tokens(self, text: str) -> int:
    encoding = tiktoken.encoding_for_model(llm_config.chat_model)
    return len(encoding.encode(text))
```

**问题**：`tiktoken.encoding_for_model()` 只支持 OpenAI 官方模型名（如 `gpt-4o`、`text-embedding-3-large`），不支持自定义模型名（如 `qwen3.7-plus`、`doubao-seed-2-1-pro-260628`）。

**修复方案**：

```python
def count_tokens(self, text: str) -> int:
    try:
        encoding = tiktoken.encoding_for_model(llm_config.chat_model)
    except KeyError:
        # 降级：使用 cl100k_base 编码（大多数现代模型使用）
        encoding = tiktoken.get_encoding("cl100k_base")
    return len(encoding.encode(text))
```

**原理**：tiktoken 支持的编码类型：
- `cl100k_base`：GPT-4、GPT-3.5-turbo、text-embedding-3 系列
- `r50k_base`：GPT-3 系列
- `p50k_base`：Codex 系列

**1. `ingest.py` 增量更新逻辑**

当前实现（`rag/ingest.py`）是全量重建，每次清空再重新入库。需要改为增量更新：

```python
import os
import json
import hashlib

INDEX_META_FILE = os.path.join(rag_config.vector_store_dir, "index_meta.json")

def get_chunk_id(chunk) -> str:
    """生成 chunk 的唯一标识"""
    content = chunk.page_content
    full_path = chunk.metadata.get("full_path", "")
    page = chunk.metadata.get("page", "")
    
    hash_str = f"{full_path}|{page}|{content[:500]}"
    return hashlib.md5(hash_str.encode("utf-8")).hexdigest()

def ingest_documents(root_dir: str = None, incremental: bool = True):
    logger.info("开始文档入库...")
    
    docs = load_all_files(root_dir)
    logger.info(f"共加载 {len(docs)} 个文档")
    
    chunks = split_documents(docs, rag_config.chunk_size, rag_config.chunk_overlap)
    logger.info(f"切分为 {len(chunks)} 个 chunk")
    
    clean_chunks = clean_documents(chunks)
    
    if incremental:
        # 增量更新模式
        index_meta = load_index_meta()
        new_chunks = []
        
        for chunk in clean_chunks:
            chunk_id = get_chunk_id(chunk)
            if chunk_id not in index_meta:
                index_meta[chunk_id] = {
                    "file_path": chunk.metadata.get("full_path", ""),
                    "file_name": chunk.metadata.get("source", ""),
                    "page": chunk.metadata.get("page", None)
                }
                new_chunks.append(chunk)
        
        if new_chunks:
            logger.info(f"增量更新: {len(new_chunks)} 个新/变更的 chunk")
            vectorStore.add_documents(new_chunks)
        else:
            logger.info("没有需要更新的文档")
        
        save_index_meta(index_meta)
    else:
        # 全量重建模式
        logger.info("全量重建索引...")
        vectorStore.clear()
        vectorStore.add_documents(clean_chunks)
    
    count = vectorStore.chroma._collection.count()
    logger.info(f"入库完成，向量库中共有 {count} 条数据")
```

**2. `chain.py` 检索效率优化**

当前 `run` 方法中 retriever 被调用了两次（一次调试，一次在 chain 中）：

```python
def run(self, question: str) -> str:
    try:
        logger.info(f"开始执行问答链，问题：{question}")
        
        # ❌ 这里调用了一次
        docs = self.retriever.invoke(question)
        logger.info(f"检索到 {len(docs)} 个相关文档")
        
        # ❌ 这里又调用了一次（在 chain 内部）
        return self.chain.invoke(question)
    except Exception as e:
        # ...
```

**优化方案**：使用 `RunnablePassthrough` 传递检索结果，避免重复调用：

```python
def run(self, question: str) -> str:
    try:
        logger.info(f"开始执行问答链，问题：{question}")
        
        # 先检索，记录日志
        docs = self.retriever.invoke(question)
        logger.info(f"检索到 {len(docs)} 个相关文档")
        
        # 使用已检索的结果，避免重复调用
        result = self.prompt.invoke({
            "context": format_docs(docs),
            "question": question
        })
        response = self.client.invoke(result.to_messages())
        
        return response.content
    except Exception as e:
        # ...
```

#### 任务 3.3：用 LangGraph 实现「任务分类 → 路由到不同 Chain」

**步骤 1**：定义状态

```python
from typing import TypedDict, Annotated
from langchain_core.messages import BaseMessage
from langgraph.graph import StateGraph, END

class AgentState(TypedDict):
    messages: Annotated[list[BaseMessage], "对话历史"]
    task_type: Annotated[str, "任务类型：rag | chat | code | translate"]
    result: Annotated[str, "最终结果"]
```

**步骤 2**：定义节点

```python
from llm_client import llm
from rag.chain import RAGChain
from typing import Literal

# 意图识别节点
def identify_intent(state: AgentState) -> AgentState:
    last_message = state["messages"][-1].content
    
    # 使用结构化输出判断任务类型
    class TaskType(BaseModel):
        type: Literal["rag", "chat", "code", "translate"]
        reason: str
    
    result = llm.chat_with_structure(
        f"判断用户问题的类型：{last_message}",
        schema=TaskType,
        system_content="请判断用户问题的类型：rag(知识库问答)、chat(普通对话)、code(代码生成)、translate(翻译)"
    )
    
    return {"task_type": result.type, "reason": result.reason}

# RAG 问答节点
def rag_handler(state: AgentState) -> AgentState:
    question = state["messages"][-1].content
    rag_chain = RAGChain()
    result = rag_chain.run(question)
    return {"result": result}

# 普通对话节点
def chat_handler(state: AgentState) -> AgentState:
    question = state["messages"][-1].content
    result = llm.chat(question)
    return {"result": result}

# 代码生成节点
def code_handler(state: AgentState) -> AgentState:
    question = state["messages"][-1].content
    result = llm.chat(question, system_content="你是专业的代码助手，请生成高质量的代码")
    return {"result": result}

# 翻译节点
def translate_handler(state: AgentState) -> AgentState:
    question = state["messages"][-1].content
    result = llm.chat(question, system_content="你是专业的翻译助手，请准确翻译用户的内容")
    return {"result": result}
```

**步骤 3**：定义路由和图

```python
# 路由函数
def route_task(state: AgentState) -> str:
    task_type = state["task_type"]
    if task_type == "rag":
        return "rag_handler"
    elif task_type == "chat":
        return "chat_handler"
    elif task_type == "code":
        return "code_handler"
    elif task_type == "translate":
        return "translate_handler"
    else:
        return "chat_handler"

# 创建图
workflow = StateGraph(AgentState)

# 添加节点
workflow.add_node("identify_intent", identify_intent)
workflow.add_node("rag_handler", rag_handler)
workflow.add_node("chat_handler", chat_handler)
workflow.add_node("code_handler", code_handler)
workflow.add_node("translate_handler", translate_handler)

# 添加边
workflow.set_entry_point("identify_intent")
workflow.add_conditional_edges("identify_intent", route_task)
workflow.add_edge("rag_handler", END)
workflow.add_edge("chat_handler", END)
workflow.add_edge("code_handler", END)
workflow.add_edge("translate_handler", END)

# 编译
app = workflow.compile()
```

#### 任务 4.2：实现 `/chat/stream` SSE 流式接口

**步骤 1**：定义请求/响应模型（`api/schemas/chat.py`）

```python
from pydantic import BaseModel

class ChatRequest(BaseModel):
    message: str
    system_content: str = None

    content: str
    finish_reason: str = None
```

**步骤 2**：实现路由（`api/routers/chat.py`）

```python
from fastapi import APIRouter, Depends
from fastapi.responses import StreamingResponse
from api.schemas.chat import ChatRequest
from llm_client import llm

router = APIRouter()

@router.post("/")
def chat(request: ChatRequest):
    """同步对话接口"""
    result = llm.chat(request.message, request.system_content)
    return {"content": result}

@router.post("/stream")
def chat_stream(request: ChatRequest):
    """流式对话接口"""
    def generate():
        for chunk in llm.stream(request.message, request.system_content):
            yield f"data: {chunk}\n\n"
    
    return StreamingResponse(generate(), media_type="text/event-stream")
```
